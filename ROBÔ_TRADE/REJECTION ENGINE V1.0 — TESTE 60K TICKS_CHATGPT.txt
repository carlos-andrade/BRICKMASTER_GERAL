input
    Janela(5);

    // ==============================
    // LIMITES PROVISÓRIOS
    // ==============================
    MinRI(30.0);
    MinWick(25.0);
    MinEvidence(30.0);

    // Progressão
    MinSlope(2.0);
    MinAcceleration(1.0);

    // Persistência
    MinPersistence(4);

    // Volume
    VolumeMedia(20);
    MinVolumeRatio(1.20);

    // Delta
    MinDeltaDivergence(0.0);

    // Exibição
    MostrarDE(true);
    MostrarRI(true);
    MostrarURE(true);
    MostrarLRE(true);
    MostrarEstados(true);
    MostrarSinais(true);


var

    // ==============================
    // PREÇO
    // ==============================
    Rng             : Float;
    Body            : Float;
    Disp            : Float;

    DE              : Float;
    RI              : Float;

    UpperWick       : Float;
    LowerWick       : Float;

    UpperWickPct    : Float;
    LowerWickPct    : Float;

    CloseLocation   : Float;

    HighFailure     : Float;
    LowFailure      : Float;

    URE             : Float;
    LRE             : Float;

    // ==============================
    // PROGRESSÃO
    // ==============================
    UREPrev         : Float;
    LREPrev         : Float;

    URP             : Float;
    LRP             : Float;

    URPPrev         : Float;
    LRPPrev         : Float;

    URSlope         : Float;
    LRSlope         : Float;

    URAccel         : Float;
    LRAccel         : Float;

    // ==============================
    // PERSISTÊNCIA
    // ==============================
    CountUpper      : Integer;
    CountLower      : Integer;

    PersistentUpper : Boolean;
    PersistentLower : Boolean;

    // ==============================
    // EXTREME REJECTION DIVERGENCE
    // ==============================
    ERDUp           : Boolean;
    ERDDown         : Boolean;

    // ==============================
    // VOLUME
    // ==============================
    VolMedia        : Float;
    VolumeRatio     : Float;

    // ==============================
    // DELTA
    // ==============================
    DeltaAtual      : Float;
    DeltaAnterior   : Float;
    DeltaChange     : Float;

    PriceChange     : Float;
    FlowDivergence  : Integer;

    // ==============================
    // ESTADOS
    // ==============================
    State           : Integer;

    UpperRejection  : Boolean;
    LowerRejection  : Boolean;

    Progressive     : Boolean;
    Persistent      : Boolean;
    FlowConfirmed   : Boolean;

    ReversalCandidate : Boolean;

    // ==============================
    // CONTROLE
    // ==============================
    i               : Integer;

begin

    // ============================================================
    // PROTEÇÃO
    // ============================================================

    if CurrentBar > Janela then
    begin

        // ========================================================
        // 1. ESTRUTURA DO CANDLE
        // ========================================================

        Rng  := High - Low;
        Body := Abs(Close - Open);
        Disp := Close - Open;

        if Rng > 0 then
        begin

            // Eficiência direcional
            DE := (Body / Rng) * 100.0;

            // Ineficiência / potencial de rejeição
            RI := 100.0 - DE;

            // Wick superior
            UpperWick := High - Max(Open, Close);

            // Wick inferior
            LowerWick := Min(Open, Close) - Low;

            UpperWickPct := (UpperWick / Rng) * 100.0;
            LowerWickPct := (LowerWick / Rng) * 100.0;

            // ====================================================
            // 2. CLOSE LOCATION
            // ====================================================

            CloseLocation := ((Close - Low) / Rng) * 100.0;

            // Falha em sustentar o topo
            HighFailure := 100.0 - CloseLocation;

            // Recuperação da mínima
            LowFailure := CloseLocation;

            // ====================================================
            // 3. EVIDÊNCIA DE REJEIÇÃO
            // ====================================================

            // Rejeição superior:
            // Wick superior + fechamento afastado do topo
            URE := UpperWickPct * (HighFailure / 100.0);

            // Rejeição inferior:
            // Wick inferior + fechamento afastado da mínima
            LRE := LowerWickPct * (LowFailure / 100.0);

        end
        else
        begin

            DE := 0;
            RI := 0;

            UpperWick := 0;
            LowerWick := 0;

            UpperWickPct := 0;
            LowerWickPct := 0;

            CloseLocation := 50;
            HighFailure := 50;
            LowFailure := 50;

            URE := 0;
            LRE := 0;

        end;


        // ========================================================
        // 4. PROGRESSÃO
        // ========================================================

        UREPrev := URE[1];
        LREPrev := LRE[1];

        URP := URE - UREPrev;
        LRP := LRE - LREPrev;

        URPPrev := URE[1] - URE[2];
        LRPPrev := LRE[1] - LRE[2];

        // Inclinação simplificada da janela
        URSlope := (URE - URE[Janela]) / Janela;
        LRSlope := (LRE - LRE[Janela]) / Janela;

        // Aceleração
        URAccel := URP - URPPrev;
        LRAccel := LRP - LRPPrev;


        // ========================================================
        // 5. PERSISTÊNCIA
        // ========================================================

        CountUpper := 0;
        CountLower := 0;

        for i := 0 to Janela - 1 do
        begin

            if URE[i] >= MinEvidence then
                CountUpper := CountUpper + 1;

            if LRE[i] >= MinEvidence then
                CountLower := CountLower + 1;

        end;

        PersistentUpper := CountUpper >= MinPersistence;
        PersistentLower := CountLower >= MinPersistence;


        // ========================================================
        // 6. EXTREME REJECTION DIVERGENCE
        // ========================================================

        ERDUp := false;
        ERDDown := false;

        // Preço faz novo topo,
        // mas rejeição superior aumenta
        // e eficiência piora.

        if CurrentBar > 1 then
        begin

            if
                (High > High[1]) and
                (URE > URE[1]) and
                (DE < DE[1])
            then
                ERDUp := true;


            // Preço faz nova mínima,
            // mas rejeição inferior aumenta
            // e eficiência piora.

            if
                (Low < Low[1]) and
                (LRE > LRE[1]) and
                (DE < DE[1])
            then
                ERDDown := true;

        end;


        // ========================================================
        // 7. VOLUME RELATIVO
        // ========================================================

        VolMedia := MediaExp(VolumeMedia, Volume);

        if VolMedia > 0 then
            VolumeRatio := Volume / VolMedia
        else
            VolumeRatio := 1.0;


        // ========================================================
        // 8. DELTA DE AGRESSÃO
        // ========================================================

        DeltaAtual := AgressionVolBalance;

        DeltaAnterior := DeltaAtual[1];

        DeltaChange := DeltaAtual - DeltaAnterior;

        PriceChange := Close - Close[1];

        FlowDivergence := 0;


        // Preço sobe e delta piora
        if
            (PriceChange > 0) and
            (DeltaChange < MinDeltaDivergence)
        then
            FlowDivergence := -1;


        // Preço cai e delta melhora
        if
            (PriceChange < 0) and
            (DeltaChange > MinDeltaDivergence)
        then
            FlowDivergence := 1;


        // ========================================================
        // 9. CLASSIFICAÇÃO PRIMÁRIA
        // ========================================================

        UpperRejection := false;
        LowerRejection := false;

        // ==========================================
        // REJEIÇÃO SUPERIOR
        // ==========================================

        if
            (RI >= MinRI) and
            (UpperWickPct >= MinWick) and
            (URE >= MinEvidence)
        then
            UpperRejection := true;


        // ==========================================
        // REJEIÇÃO INFERIOR
        // ==========================================

        if
            (RI >= MinRI) and
            (LowerWickPct >= MinWick) and
            (LRE >= MinEvidence)
        then
            LowerRejection := true;


        // ========================================================
        // 10. PROGRESSÃO
        // ========================================================

        Progressive := false;

        if
            (
                (UpperRejection and
                 (URSlope >= MinSlope) and
                 (URAccel >= MinAcceleration))
                or
                (LowerRejection and
                 (LRSlope >= MinSlope) and
                 (LRAccel >= MinAcceleration))
            )
        then
            Progressive := true;


        // ========================================================
        // 11. PERSISTÊNCIA
        // ========================================================

        Persistent := false;

        if
            (
                (UpperRejection and PersistentUpper)
                or
                (LowerRejection and PersistentLower)
            )
        then
            Persistent := true;


        // ========================================================
        // 12. CONFIRMAÇÃO DE FLUXO
        // ========================================================

        FlowConfirmed := false;

        // --------------------------------------------------------
        // REJEIÇÃO SUPERIOR + fluxo vendedor
        // --------------------------------------------------------

        if
            UpperRejection and
            (
                (FlowDivergence = -1)
                or
                (DeltaAtual < DeltaAnterior)
            ) and
            (VolumeRatio >= MinVolumeRatio)
        then
            FlowConfirmed := true;


        // --------------------------------------------------------
        // REJEIÇÃO INFERIOR + fluxo comprador
        // --------------------------------------------------------

        if
            LowerRejection and
            (
                (FlowDivergence = 1)
                or
                (DeltaAtual > DeltaAnterior)
            ) and
            (VolumeRatio >= MinVolumeRatio)
        then
            FlowConfirmed := true;


        // ========================================================
        // 13. MÁQUINA DE ESTADOS
        // ========================================================

        State := 0;


        // --------------------------------------------------------
        // STATE 1
        // PERDA DE EFICIÊNCIA
        // --------------------------------------------------------

        if RI >= MinRI then
            State := 1;


        // --------------------------------------------------------
        // STATE 2
        // REJEIÇÃO EM DESENVOLVIMENTO
        // --------------------------------------------------------

        if
            (
                UpperRejection or
                LowerRejection
            )
        then
            State := 2;


        // --------------------------------------------------------
        // STATE 3
        // REJEIÇÃO ATIVA
        // --------------------------------------------------------

        if
            (
                UpperRejection or
                LowerRejection
            ) and
            (URE >= MinEvidence or LRE >= MinEvidence)
        then
            State := 3;


        // --------------------------------------------------------
        // STATE 4
        // REJEIÇÃO PROGRESSIVA
        // --------------------------------------------------------

        if Progressive then
            State := 4;


        // --------------------------------------------------------
        // STATE 5
        // REJEIÇÃO PERSISTENTE
        // --------------------------------------------------------

        if Persistent then
            State := 5;


        // --------------------------------------------------------
        // STATE 6
        // FLUXO CONFIRMADO
        // --------------------------------------------------------

        if FlowConfirmed then
            State := 6;


        // ========================================================
        // 14. CANDIDATO A REVERSÃO
        // ========================================================

        ReversalCandidate := false;


        // --------------------------------------------------------
        // REVERSÃO PARA BAIXO
        // Rejeição superior
        // + progressão
        // + persistência
        // + fluxo
        // --------------------------------------------------------

        if
            UpperRejection and
            Progressive and
            Persistent and
            FlowConfirmed
        then
            ReversalCandidate := true;


        // --------------------------------------------------------
        // REVERSÃO PARA CIMA
        // Rejeição inferior
        // + progressão
        // + persistência
        // + fluxo
        // --------------------------------------------------------

        if
            LowerRejection and
            Progressive and
            Persistent and
            FlowConfirmed
        then
            ReversalCandidate := true;


        // ========================================================
        // STATE 7
        // REVERSÃO CANDIDATA
        // ========================================================

        if ReversalCandidate then
            State := 7;


        // ========================================================
        // 15. PLOTS
        // ========================================================

        if MostrarDE then
            Plot(DE)
        else
            NoPlot(1);


        if MostrarRI then
            Plot2(RI)
        else
            NoPlot(2);


        if MostrarURE then
            Plot3(URE)
        else
            NoPlot(3);


        if MostrarLRE then
            Plot4(LRE)
        else
            NoPlot(4);


        if MostrarEstados then
            Plot5(State)
        else
            NoPlot(5);


        // ========================================================
        // 16. MARCAÇÕES NO GRÁFICO
        // ========================================================

        if MostrarSinais then
        begin

            // ----------------------------------------------------
            // REJEIÇÃO SUPERIOR
            // ----------------------------------------------------

            if UpperRejection then
                PlotText("REJ SUP", 0, High + (Rng * 0.15));


            // ----------------------------------------------------
            // REJEIÇÃO INFERIOR
            // ----------------------------------------------------

            if LowerRejection then
                PlotText("REJ INF", 0, Low - (Rng * 0.15));


            // ----------------------------------------------------
            // ERD SUPERIOR
            // ----------------------------------------------------

            if ERDUp then
                PlotText("ERD UP", 0, High + (Rng * 0.30));


            // ----------------------------------------------------
            // ERD INFERIOR
            // ----------------------------------------------------

            if ERDDown then
                PlotText("ERD DOWN", 0, Low - (Rng * 0.30));


            // ----------------------------------------------------
            // FLUXO CONFIRMADO
            // ----------------------------------------------------

            if FlowConfirmed then
                PlotText("FLOW", 0, Close);


            // ----------------------------------------------------
            // REVERSÃO CANDIDATA
            // ----------------------------------------------------

            if ReversalCandidate then
                PlotText("REVERSAO", 0, Close);

        end;

    end;

end;
