// ============================================================================
// REJECTION ENGINE — V1.0.1
// ----------------------------------------------------------------------------
// BRICKMASTER — MÓDULO EXPERIMENTAL INDEPENDENTE
// Plataforma: Profit / Nelogica
// Linguagem: NTSL
// Gráfico de referência: 60.000 TICKS
//
// OBJETIVO:
// Detectar perda de eficiência direcional e evidências de rejeição
// nos extremos do candle.
//
// ESTA VERSÃO É EXCLUSIVAMENTE DIAGNÓSTICA.
// NÃO EXECUTA ORDENS.
//
// MÓDULOS PRESENTES:
// 1. Eficiência Direcional
// 2. Ineficiência / Potencial de Rejeição
// 3. Wick Superior / Inferior
// 4. Close Location
// 5. Upper Rejection Evidence
// 6. Lower Rejection Evidence
// 7. Progressão
// 8. Persistência
// 9. Estado
//
// FORA DESTA VERSÃO:
// VIX / FVG / VWAP / RSI / MACD / AgressionVol / CVD / EXECUÇÃO
// ============================================================================

input

  // --- Parâmetros principais ---
  MinRI(30.0);
  MinWick(25.0);
  MinEvidence(30.0);

  // --- Progressão ---
  MinSlope(2.0);

  // --- Persistência ---
  MinPersistence(4);


var

  // ========================================================================
  // DADOS DO CANDLE
  // ========================================================================

  RangeCandle        : Float;
  Displacement       : Float;
  Body               : Float;

  // ========================================================================
  // EFICIÊNCIA
  // ========================================================================

  DirectionalEfficiency : Float;
  RejectionInefficiency : Float;

  // ========================================================================
  // WICKS
  // ========================================================================

  UpperWick           : Float;
  LowerWick           : Float;

  UpperWickPercent     : Float;
  LowerWickPercent     : Float;

  // ========================================================================
  // POSIÇÃO DO FECHAMENTO
  // ========================================================================

  CloseLocation       : Float;
  HighFailure         : Float;
  LowFailure          : Float;

  // ========================================================================
  // EVIDÊNCIA DE REJEIÇÃO
  // ========================================================================

  UpperRejectionEvidence : Float;
  LowerRejectionEvidence : Float;

  // ========================================================================
  // PROGRESSÃO
  // ========================================================================

  UpperSlope          : Float;
  LowerSlope          : Float;

  UpperProgression    : Float;
  LowerProgression    : Float;

  // ========================================================================
  // PERSISTÊNCIA
  // ========================================================================

  UpperPersistence    : Integer;
  LowerPersistence    : Integer;

  // ========================================================================
  // DIREÇÃO
  // ========================================================================

  UpperRejection      : Boolean;
  LowerRejection      : Boolean;

  // ========================================================================
  // ESTADO
  // ========================================================================

  RejectionState      : Integer;


begin

  // ==========================================================================
  // 1. RANGE
  // ==========================================================================

  RangeCandle := High - Low;


  // ==========================================================================
  // 2. DISPLACEMENT E BODY
  // ==========================================================================

  Displacement := Close - Open;

  Body := Abs(Close - Open);


  // ==========================================================================
  // 3. CÁLCULOS PRINCIPAIS
  // ==========================================================================

  if RangeCandle > 0 then
  begin

    // ------------------------------------------------------------------------
    // EFICIÊNCIA DIRECIONAL
    //
    // DE = |Close - Open| / Range * 100
    // ------------------------------------------------------------------------

    DirectionalEfficiency :=
      (Body / RangeCandle) * 100.0;


    // ------------------------------------------------------------------------
    // INEFICIÊNCIA / POTENCIAL DE REJEIÇÃO
    //
    // RI = 100 - DE
    // ------------------------------------------------------------------------

    RejectionInefficiency :=
      100.0 - DirectionalEfficiency;


    // ------------------------------------------------------------------------
    // WICK SUPERIOR
    // ------------------------------------------------------------------------

    UpperWick :=
      High - Max(Open, Close);


    // ------------------------------------------------------------------------
    // WICK INFERIOR
    // ------------------------------------------------------------------------

    LowerWick :=
      Min(Open, Close) - Low;


    // ------------------------------------------------------------------------
    // WICK %
    // ------------------------------------------------------------------------

    UpperWickPercent :=
      (UpperWick / RangeCandle) * 100.0;

    LowerWickPercent :=
      (LowerWick / RangeCandle) * 100.0;


    // ------------------------------------------------------------------------
    // CLOSE LOCATION
    //
    // 0   = fechamento na mínima
    // 50  = fechamento no meio
    // 100 = fechamento na máxima
    // ------------------------------------------------------------------------

    CloseLocation :=
      ((Close - Low) / RangeCandle) * 100.0;


    // ------------------------------------------------------------------------
    // HIGH FAILURE
    //
    // Quanto mais alto, mais distante o fechamento ficou da máxima.
    // ------------------------------------------------------------------------

    HighFailure :=
      100.0 - CloseLocation;


    // ------------------------------------------------------------------------
    // LOW FAILURE
    //
    // Quanto mais alto, mais distante o fechamento ficou da mínima.
    // ------------------------------------------------------------------------

    LowFailure :=
      CloseLocation;


    // ------------------------------------------------------------------------
    // UPPER REJECTION EVIDENCE
    //
    // Rejeição da região superior.
    //
    // Wick superior grande +
    // fechamento distante da máxima.
    // ------------------------------------------------------------------------

    UpperRejectionEvidence :=
      UpperWickPercent *
      (HighFailure / 100.0);


    // ------------------------------------------------------------------------
    // LOWER REJECTION EVIDENCE
    //
    // Rejeição da região inferior.
    //
    // Wick inferior grande +
    // fechamento distante da mínima.
    // ------------------------------------------------------------------------

    LowerRejectionEvidence :=
      LowerWickPercent *
      (LowFailure / 100.0);

  end
  else
  begin

    DirectionalEfficiency := 0.0;
    RejectionInefficiency := 0.0;

    UpperWick := 0.0;
    LowerWick := 0.0;

    UpperWickPercent := 0.0;
    LowerWickPercent := 0.0;

    CloseLocation := 50.0;

    HighFailure := 50.0;
    LowFailure := 50.0;

    UpperRejectionEvidence := 0.0;
    LowerRejectionEvidence := 0.0;

  end;


  // ==========================================================================
  // 4. PROGRESSÃO
  //
  // Janela experimental = 5 candles de 60K.
  //
  // Não usamos tempo.
  // Não usamos minutos.
  // Cada unidade é um candle de 60.000 ticks.
  // ==========================================================================

  UpperProgression :=
    UpperRejectionEvidence -
    UpperRejectionEvidence[1];

  LowerProgression :=
    LowerRejectionEvidence -
    LowerRejectionEvidence[1];


  // ==========================================================================
  // 5. SLOPE
  //
  // Aproximação simples usando 5 candles:
  //
  // Slope = (valor atual - valor de 4 candles atrás) / 4
  // ==========================================================================

  if CurrentBar() >= 5 then
  begin

    UpperSlope :=
      (UpperRejectionEvidence -
       UpperRejectionEvidence[4]) / 4.0;

    LowerSlope :=
      (LowerRejectionEvidence -
       LowerRejectionEvidence[4]) / 4.0;

  end
  else
  begin

    UpperSlope := 0.0;
    LowerSlope := 0.0;

  end;


  // ==========================================================================
  // 6. PERSISTÊNCIA
  //
  // Conta quantos dos últimos 5 candles apresentam evidência acima
  // do limiar definido.
  // ==========================================================================

  UpperPersistence := 0;
  LowerPersistence := 0;


  if UpperRejectionEvidence >= MinEvidence then
    UpperPersistence := UpperPersistence + 1;

  if UpperRejectionEvidence[1] >= MinEvidence then
    UpperPersistence := UpperPersistence + 1;

  if UpperRejectionEvidence[2] >= MinEvidence then
    UpperPersistence := UpperPersistence + 1;

  if UpperRejectionEvidence[3] >= MinEvidence then
    UpperPersistence := UpperPersistence + 1;

  if UpperRejectionEvidence[4] >= MinEvidence then
    UpperPersistence := UpperPersistence + 1;


  if LowerRejectionEvidence >= MinEvidence then
    LowerPersistence := LowerPersistence + 1;

  if LowerRejectionEvidence[1] >= MinEvidence then
    LowerPersistence := LowerPersistence + 1;

  if LowerRejectionEvidence[2] >= MinEvidence then
    LowerPersistence := LowerPersistence + 1;

  if LowerRejectionEvidence[3] >= MinEvidence then
    LowerPersistence := LowerPersistence + 1;

  if LowerRejectionEvidence[4] >= MinEvidence then
    LowerPersistence := LowerPersistence + 1;


  // ==========================================================================
  // 7. CLASSIFICAÇÃO DA REJEIÇÃO
  // ==========================================================================

  UpperRejection := false;
  LowerRejection := false;


  // --------------------------------------------------------------------------
  // REJEIÇÃO SUPERIOR
  //
  // Wick superior relevante
  // +
  // evidência relevante
  // +
  // ineficiência relevante
  // --------------------------------------------------------------------------

  if
    (UpperWickPercent >= MinWick) and
    (UpperRejectionEvidence >= MinEvidence) and
    (RejectionInefficiency >= MinRI)
  then
    UpperRejection := true;


  // --------------------------------------------------------------------------
  // REJEIÇÃO INFERIOR
  // --------------------------------------------------------------------------

  if
    (LowerWickPercent >= MinWick) and
    (LowerRejectionEvidence >= MinEvidence) and
    (RejectionInefficiency >= MinRI)
  then
    LowerRejection := true;


  // ==========================================================================
  // 8. ESTADO
  //
  // 0 = NORMAL
  // 1 = PERDA DE EFICIÊNCIA
  // 2 = REJEIÇÃO DESENVOLVENDO
  // 3 = REJEIÇÃO ATIVA
  // 4 = REJEIÇÃO PROGRESSIVA
  // 5 = REJEIÇÃO PERSISTENTE
  // ==========================================================================

  RejectionState := 0;


  // --------------------------------------------------------------------------
  // ESTADO 1 — PERDA DE EFICIÊNCIA
  // --------------------------------------------------------------------------

  if RejectionInefficiency >= MinRI then
    RejectionState := 1;


  // --------------------------------------------------------------------------
  // ESTADO 2 — REJEIÇÃO DESENVOLVENDO
  // --------------------------------------------------------------------------

  if
    (UpperRejection = true) or
    (LowerRejection = true)
  then
    RejectionState := 2;


  // --------------------------------------------------------------------------
  // ESTADO 3 — REJEIÇÃO ATIVA
  // --------------------------------------------------------------------------

  if
    (
      (UpperRejection = true) and
      (UpperProgression > 0)
    )
    or
    (
      (LowerRejection = true) and
      (LowerProgression > 0)
    )
  then
    RejectionState := 3;


  // --------------------------------------------------------------------------
  // ESTADO 4 — REJEIÇÃO PROGRESSIVA
  // --------------------------------------------------------------------------

  if
    (
      (UpperSlope >= MinSlope) and
      (UpperRejection = true)
    )
    or
    (
      (LowerSlope >= MinSlope) and
      (LowerRejection = true)
    )
  then
    RejectionState := 4;


  // --------------------------------------------------------------------------
  // ESTADO 5 — REJEIÇÃO PERSISTENTE
  // --------------------------------------------------------------------------

  if
    (
      (UpperPersistence >= MinPersistence) and
      (UpperSlope >= MinSlope)
    )
    or
    (
      (LowerPersistence >= MinPersistence) and
      (LowerSlope >= MinSlope)
    )
  then
    RejectionState := 5;


  // ==========================================================================
  // 9. PLOTS
  //
  // Plot1 = Eficiência Direcional
  // Plot2 = Ineficiência
  // Plot3 = Upper Rejection Evidence
  // Plot4 = Lower Rejection Evidence
  // Plot5 = Upper Wick %
  // Plot6 = Lower Wick %
  // Plot7 = Estado
  // ==========================================================================

  Plot(DirectionalEfficiency);

  Plot2(RejectionInefficiency);

  Plot3(UpperRejectionEvidence);

  Plot4(LowerRejectionEvidence);

  Plot5(UpperWickPercent);

  Plot6(LowerWickPercent);

  Plot7(RejectionState);


end;
