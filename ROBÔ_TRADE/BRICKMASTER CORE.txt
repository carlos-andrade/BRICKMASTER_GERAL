// ==========================================================
// BRICKMASTER CORE - EXECUTION AWARE ENGINE
// Estrutura: Renko 15R (70 pts) + reversão 140 pts
// Integração: Slippage + Fluxo (TR, Volume, Delta, VWAP-TWAP)
// Gestão: Trailing dinâmico com WHILE
// Observabilidade: Plot + PaintBar
// ==========================================================

// =========================
// INPUTS (AJUSTÁVEIS)
// =========================
input:
    BrickSize(70),            // Tamanho do tijolo (continuação)
    ReversalSize(140),        // Tamanho da reversão
    TickSize(5),              // 1 tick (WIN = 5 pontos)
    MaxSlippage(2),           // Slippage máximo aceitável (ticks)
    LookbackTWAP(20),         // Período TWAP
    FatorAgressao(1.5),       // Peso da agressão
    FatorVWAP(1.0);           // Peso do desequilíbrio

// =========================
// VARIÁVEIS
// =========================
var:

    // -------- Estrutura Renko --------
    baseRenko(0),             // Base atual do Renko
    direcao(0),               // 1 = alta | -1 = baixa

    // -------- Execução --------
    precoAlvo(0),             // Preço teórico de entrada
    precoExecucao(0),         // Preço real executado
    slippageTicks(0),         // Slippage em ticks
    slippageAbs(0),           // Slippage absoluto

    posAtual(0),
    posAnterior(0),

    // -------- Fluxo --------
    tr(0),                    // True Range
    volumeFinanceiro(0),      // Volume financeiro
    agressao(0),              // Proxy de agressão

    delta(0),                 // Delta acumulado (proxy)
    deltaAnterior(0),

    vwap(0),
    twap(0),
    gapVWAP(0),

    scoreFluxo(0),            // Score combinado de fluxo

    // -------- Regime --------
    regimeHostil(false),      // Mercado ruim para execução

    // -------- Gestão --------
    trailingStop(0),
    ultimoTopo(0),
    ultimoFundo(0);

// ==========================================================
// BLOCO 1 — CONSTRUÇÃO RENKO (REMOVE RUÍDO)
// ==========================================================

// Inicialização da base
if (baseRenko = 0) then
    baseRenko = Close;

// Continuação da tendência (70 pontos)
if (Close >= baseRenko + BrickSize) then
begin
    baseRenko = baseRenko + BrickSize;
    direcao = 1;
end;

// Reversão forte (140 pontos)
if (Close <= baseRenko - ReversalSize) then
begin
    baseRenko = baseRenko - BrickSize;
    direcao = -1;
end;


// ==========================================================
// BLOCO 2 — LEITURA DE FLUXO (MICROESTRUTURA)
// ==========================================================

// True Range
tr = TrueRange;

// Volume financeiro (dinheiro real no mercado)
volumeFinanceiro = Volume * Close;

// Evita divisão por zero
if (tr > 0) then
    agressao = volumeFinanceiro / tr
else
    agressao = 0;

// Delta (proxy simplificado de agressão)
delta = deltaAnterior + (Close - Close[1]) * Volume;

// VWAP (institucional)
vwap = VWAP;

// TWAP (média temporal)
twap = Average(Close, LookbackTWAP);

// Gap entre VWAP e TWAP (desequilíbrio)
gapVWAP = vwap - twap;

// Score final de fluxo
scoreFluxo =
    (AbsValue(gapVWAP) * FatorVWAP) +
    (agressao * FatorAgressao);


// ==========================================================
// BLOCO 3 — DETECÇÃO DE REGIME HOSTIL
// ==========================================================

regimeHostil = false;

// Mercado com alta fricção
if (scoreFluxo > 150000) then
    regimeHostil = true;

// Desequilíbrio extremo
if (AbsValue(gapVWAP) > 50) then
    regimeHostil = true;


// ==========================================================
// BLOCO 4 — ENTRADA ADAPTATIVA (EXECUÇÃO INTELIGENTE)
// ==========================================================

// -------- COMPRA --------
if (direcao = 1) then
begin
    precoAlvo = baseRenko;

    // Confirmação por fluxo (delta crescente)
    if (delta > deltaAnterior) and (not regimeHostil) then
    begin
        // Mercado estável → agressivo (market)
        if (scoreFluxo < 80000) then
            BuyAtMarket

        // Mercado sensível → passivo (limit)
        else
            BuyLimit(precoAlvo);
    end;
end;


// -------- VENDA --------
if (direcao = -1) then
begin
    precoAlvo = baseRenko;

    if (delta < deltaAnterior) and (not regimeHostil) then
    begin
        if (scoreFluxo < 80000) then
            SellShortAtMarket
        else
            SellShortLimit(precoAlvo);
    end;
end;


// ==========================================================
// BLOCO 5 — CÁLCULO DE SLIPPAGE (REAL)
// ==========================================================

posAtual = Position;

// Detecta mudança de posição (entrada executada)
if (posAtual <> posAnterior) then
begin
    precoExecucao = AvgPrice;

    // Slippage em ticks
    slippageTicks = (precoExecucao - precoAlvo) / TickSize;

    slippageAbs = AbsValue(slippageTicks);
end;

posAnterior = posAtual;


// ==========================================================
// BLOCO 6 — PROTEÇÃO POR SLIPPAGE
// ==========================================================

// Se execução foi ruim → ativa regime hostil
if (slippageAbs > MaxSlippage) then
    regimeHostil = true;


// ==========================================================
// BLOCO 7 — TRAILING STOP DINÂMICO (WHILE)
// ==========================================================

// -------- POSIÇÃO COMPRADA --------
if (Position > 0) then
begin
    // Inicializa topo
    if (ultimoTopo = 0) then
        ultimoTopo = High;

    // Atualiza topo enquanto preço sobe
    while (Close > ultimoTopo) do
        ultimoTopo = Close;

    // Stop segue o movimento
    trailingStop = ultimoTopo - BrickSize;

    SellToCoverStop(trailingStop);
end;


// -------- POSIÇÃO VENDIDA --------
if (Position < 0) then
begin
    if (ultimoFundo = 0) then
        ultimoFundo = Low;

    while (Close < ultimoFundo) do
        ultimoFundo = Close;

    trailingStop = ultimoFundo + BrickSize;

    BuyToCoverStop(trailingStop);
end;


// ==========================================================
// BLOCO 8 — TELEMETRIA (OBSERVABILIDADE)
// ==========================================================

// Slippage → custo real
Plot1(slippageTicks);

// Pressão de mercado
Plot2(scoreFluxo);

// Direção de fluxo
Plot3(delta);

// Desequilíbrio institucional
Plot4(gapVWAP);


// ==========================================================
// BLOCO 9 — ALERTAS VISUAIS
// ==========================================================

// Mercado hostil
if (regimeHostil) then
    PaintBar(clRed);

// Execução ruim
if (slippageAbs > MaxSlippage) then
    PaintBar(clYellow);


// ==========================================================
// BLOCO FINAL — ATUALIZAÇÕES DE ESTADO
// ==========================================================

// Atualiza delta para próximo ciclo
deltaAnterior = delta;
